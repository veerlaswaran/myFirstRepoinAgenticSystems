#Create the Table
CREATE TABLE students (
    id INTEGER PRIMARY KEY,
    name TEXT,
    score INTEGER,
    passed BOOLEAN,
    class INTEGER
);

#Retrieve all columns from the students table.
SELECT * FROM students;

#Retrieve only the name and score columns.
SELECT name, score FROM students;

#Retrieve students who scored above 80.
SELECT * FROM students
WHERE score > 80;

#Retrieve students who scored above 80 AND passed.
SELECT * FROM students
WHERE score > 80 AND passed = TRUE;


#Retrieve students who scored above 85 OR belong to class 9.
SELECT * FROM students
WHERE score > 85 OR class = 9;

#Retrieve students who did NOT pass the exam.(Use the NOT operator.)
SELECT * FROM students
WHERE NOT passed;

#Return a list of distinct classes.
SELECT DISTINCT class FROM students;
 

#Retrieve the top 5 highest-scoring students.
SELECT * FROM students
ORDER BY score DESC
LIMIT 5;


#Retrieve students sorted first by class (ascending) and then by score (descending).
SELECT * FROM students
ORDER BY class ASC, score DESC;


# Use an alias to rename score as final_score.
SELECT name, score AS final_score FROM students;
